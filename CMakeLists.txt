  target_compile_options(rv_dsp_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rv_dsp_tests PRIVATE -fsanitize=address,undefined)

endif()
if(RV_BUILD_PLUGINS)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/JUCE/CMakeLists.txt")
  add_subdirectory(.deps/JUCE EXCLUDE_FROM_ALL)
 else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)

 juce_add_binary_data(RVHardwareAssets HEADER_NAME SkinData.h NAMESPACE SkinData
  SOURCES Assets/Hardware/chassis.png Assets/Hardware/knob.png Assets/Hardware/button.png
          Assets/Hardware/rack.png Assets/Hardware/glass.png)
 set(RV_SOURCES Source/Plugin/PluginProcessor.cpp Source/UI/PluginEditor.cpp Source/Presets/PresetManager.cpp)
 function(rv_plugin target product code auType formats midiEdition)
  juce_add_plugin(${target}
   COMPANY_NAME RSTK BUNDLE_ID "audio.rstk.${target}" PLUGIN_MANUFACTURER_CODE Rstk
   PLUGIN_CODE ${code} PRODUCT_NAME "${product}" DESCRIPTION "Creative filter-bank vocoder"
   FORMATS ${formats} AU_MAIN_TYPE ${auType} IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE
   NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE
   COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Modulation
   MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "R-Vocoder uses audio input to process your voice.")
  target_sources(${target} PRIVATE ${RV_SOURCES})
  target_include_directories(${target} PUBLIC Source)
  target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 RV_MIDI_EDITION=${midiEdition})
  target_link_libraries(${target} PRIVATE rv_dsp juce::juce_audio_utils juce::juce_dsp PUBLIC RVHardwareAssets juce::juce_recommended_config_flags)
 endfunction()
